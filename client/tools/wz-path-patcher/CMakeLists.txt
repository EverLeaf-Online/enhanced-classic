cmake_minimum_required(VERSION 3.16)
project(everleaf_wz_path_patcher LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED LIBWZ_SOURCE)
  message(FATAL_ERROR "LIBWZ_SOURCE is required")
endif()
add_subdirectory("${LIBWZ_SOURCE}" "${CMAKE_BINARY_DIR}/libwz" EXCLUDE_FROM_ALL)
add_executable(everleaf-wz-path-patcher main.cpp)
target_link_libraries(everleaf-wz-path-patcher PRIVATE wz)
